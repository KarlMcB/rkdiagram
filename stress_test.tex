% ============================================================
%  stress_test.tex
%  Stress test for the rkdiagram LaTeX package:
%  the Nabokov "dresser with mirrors" sentence ("The Leonardo", 1933).
%
%  Fixes applied relative to the first attempt in rk-test.tex:
%    1. Page overflow  -> A3 landscape + \resizebox to text width.
%    2. Per-branch objects on the compound predicate -> the sentence is
%       split with \rkCompoundSent, so "turned" keeps "his head" and
%       "saw" keeps "parallelogram".
%    3. Branch modifier collisions -> each clause now lays out on its
%       own baseline, so the two modifier groups cannot overlap.
%    4. Three-item series ("this sky, these boughs, this gliding
%       facade") -> two items on the compound, third carried as an
%       appositive (the package supports two-branch compounds only).
%
%  Remaining known limitations (package-level):
%    - \rkAppos is text-only, so the "across which ..." relative clause
%      attaches to "parallelogram" rather than to "dresser".
%
%  Compile with: latex stress_test.tex && dvipdf stress_test.dvi
% ============================================================

\documentclass[12pt]{article}

\usepackage[
  spacing   = 1.1cm,
  modDepth  = 0.9cm,
  font      = \rmfamily,
  linewidth = 0.5pt,
  color     = black
]{rkdiagram}

\usepackage[a3paper,landscape,margin=1.5cm]{geometry}
\usepackage{graphicx}
\usepackage{parskip}

\begin{document}

\begin{center}
  {\LARGE\bfseries \texttt{rkdiagram} Stress Test}\\[4pt]
  {\large Vladimir Nabokov, ``The Leonardo'' (1933)}
\end{center}

\bigskip

\noindent\textit{``As he crossed toward the pharmacy at the corner he
involuntarily turned his head because of a burst of light that had
ricocheted from his temple, and saw, with that quick smile with which we
greet a rainbow or a rose, a blindingly white parallelogram of sky being
unloaded from the van---a dresser with mirrors across which, as across a
cinema screen, passed a flawlessly clear reflection of boughs sliding and
swaying not arboreally, but with a human vacillation, produced by the
nature of those who were carrying this sky, these boughs, this gliding
fa\c{c}ade.''}

\bigskip

\noindent\resizebox{\textwidth}{!}{%
  \rkCompoundSent[and][13cm]%
    [spacing=1.4cm,modDepth=1.0cm]%
    {%
      % ---- Clause A: "As he crossed toward the pharmacy at the corner
      %      he involuntarily turned his head because of a burst of
      %      light that had ricocheted from his temple" ----
      \begin{rkdiagram}
        \rkSubject{he}
        \rkVerb[
          \rkAdv{involuntarily},
          \rkPrep[
            \rkAdj{a},
            \rkPrep[
              \rkSubClause[that]{%
                \rkVerb[
                  \rkPrep[
                    \rkAdj{his}
                  ]{from}{temple}%
                ]{had~ricocheted}%
              }%
            ]{of}{light}%
          ]{because~of}{burst},
          \rkset{subClauseDepth=8.5cm}
          \rkSubClause[As]{%
            \rkSubject{he}%
            \rkVerb[
              \rkPrep[
                \rkAdj{the},
                \rkPrep[
                  \rkAdj{the}
                ]{at}{corner}%
              ]{toward}{pharmacy}%
            ]{crossed}%
          }%
        ]{turned}
        \rkDObj[
          \rkAdj{his}
        ]{head}
      \end{rkdiagram}%
    }%
    [spacing=1.1cm,modDepth=0.9cm]%
    {%
      % ---- Clause B: "(he) saw, with that quick smile with which we
      %      greet a rainbow or a rose, a blindingly white parallelogram
      %      of sky being unloaded from the van---a dresser with mirrors
      %      across which, as across a cinema screen, passed a
      %      flawlessly clear reflection of boughs sliding and swaying
      %      not arboreally, but with a human vacillation, produced by
      %      the nature of those who were carrying this sky, these
      %      boughs, this gliding facade" ----
      \begin{rkdiagram}
        \rkSubject{(he)}
        \rkVerb[
          \rkPrep[
            \rkAdj{that},
            \rkAdj{quick},
            \rkset{subClauseDepth=6cm}
            \rkSubClause[with~which]{%
              \rkSubject{we}%
              \rkVerb{greet}%
              \rkDObj{%
                \rkCompound[or][
                  \rkAdj{a}
                ]{rainbow}[
                  \rkAdj{a}
                ]{rose}%
              }%
            }%
          ]{with}{smile}%
        ]{saw}
        \rkDObj[
          \rkAdj{a},
          \rkAdj[
            \rkAdv{blindingly}
          ]{white},
          \rkPrep{of}{sky},
          \rkParticiple[
            \rkPrep[
              \rkAdj{the}
            ]{from}{van}%
          ]{being~unloaded},
          \rkAppos{a dresser with mirrors},
          \rkset{subClauseDepth=2.5cm}
          \rkSubClause[across~which]{%
            \rkSubject[
              \rkAdj{a},
              \rkAdj[
                \rkAdv{flawlessly}
              ]{clear},
              \rkPrep[
                \rkParticiple[
                  \rkAdv{not~arboreally},
                  \rkPrep[
                    \rkAdj{a},
                    \rkAdj{human},
                    \rkParticiple[
                      \rkPrep[
                        \rkAdj{the},
                        \rkPrep[
                          \rkSubClause[who]{%
                            \rkVerb{were~carrying}%
                            \rkDObj[
                              \rkAppos{these boughs}
                            ]{%
                              \rkCompound[and][
                                \rkAdj{this}
                              ]{sky}[
                                \rkAdj{this},
                                \rkAdj{gliding}
                              ]{fa\c{c}ade}%
                            }%
                          }%
                        ]{of}{those}%
                      ]{by}{nature}%
                    ]{produced}%
                  ]{but~with}{vacillation}%
                ]{sliding~and~swaying}%
              ]{of}{boughs}%
            ]{reflection}
            \rkVerb[
              \rkPrep[
                \rkAdj{a},
                \rkAdj{cinema}
              ]{as~across}{screen}%
            ]{passed}%
          }%
        ]{parallelogram}
      \end{rkdiagram}%
    }%
}

\bigskip

\noindent\textit{Structural notes: compound sentence join carries the
two predicates (giving each verb its own object); ``As'' adverbial
clause and ``because of'' chain on \textbf{turned}; ``with that quick
smile'' (with its own relative clause and compound object) on
\textbf{saw}; the dresser appositive is text-only, so the ``across
which'' relative clause hangs from \textbf{parallelogram}; the
three-item series is rendered as a two-branch compound plus an
appositive.}
