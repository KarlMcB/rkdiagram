\documentclass[12pt]{article}

% ---- Package under test ------------------------------------
\usepackage[
spacing   = 1.3cm,
modDepth  = 1.0cm,
font      = \rmfamily,
linewidth = 0.5pt,
color     = black
]{rkdiagram}

% ---- Support packages --------------------------------------
\usepackage[a3paper]{geometry}
\geometry{margin=1in}
\usepackage{booktabs}
\usepackage{microtype}
\usepackage{parskip}
\usepackage{xcolor}

% ---- Document styling --------------------------------------
\newcommand{\testsection}[1]{%
	\bigskip
	\noindent\rule{\linewidth}{0.8pt}\par\vspace{2pt}%
	{\noindent\raggedright\large\bfseries #1\par}\vspace{4pt}%
}

\newcommand{\testcase}[2]{%
	\medskip
	{\noindent\raggedright\textbf{Test \thesubsection:} \textit{``#1''}\par}%
	\stepcounter{subsection}\smallskip
	#2
	\par\medskip
}

% Manual counter for test numbering (subsection already defined by article class)
\setcounter{subsection}{1}

% ============================================================
\begin{document}
	

	
	% ============================================================
	\testsection{Group 12 — Mark Twain Stress Test}
	% ============================================================
	
	% Compound sentence (two independent clauses joined by ``and'').
	% Diagrammed as two stacked baselines.
	\testcase{A swarm of swarthy, noisy, lying, shoulder-shrugging, gesticulating
		Portuguese boatmen, with brass rings in their ears and fraud in their hearts,
		climbed the ship's sides, and various parties of us contracted with them to
		take us ashore at so much a head, silver coin of any country.}{%
		% ------ Clause 1: A swarm … climbed the ship's sides ------
		\begin{rkdiagram}[spacing=1.0cm,modDepth=0.9cm]
			\rkSubject[%
			\rkAdj{A},
			\rkPrep[%
			\rkAdj{swarthy},
			\rkAdj{noisy},
			\rkAdj{lying},
			\rkAdj{shoulder-shrugging},
			\rkAdj{gesticulating},
			\rkAdj{Portuguese},
			\rkPrep[%
			\rkAdj{brass},
			\rkPrep[\rkAdj{their}]{in}{ears}%
			]{with}{rings},
			\rkPrep[%
			\rkPrep[\rkAdj{their}]{in}{hearts}%
			]{with}{fraud}%
			]{of}{boatmen}%
			]{swarm}
			\rkVerb{climbed}
			\rkDObj[\rkAdj{the},\rkAdj{ship's}]{sides}
		\end{rkdiagram}
		
		\medskip
		\hspace{1.5cm}\textit{and}
		\medskip
		
		% ------ Clause 2: various parties … contracted … ------
		\begin{rkdiagram}[spacing=1.0cm,modDepth=0.9cm]
			\rkSubject[%
			\rkAdj{various},
			\rkPrep{of}{us}%
			]{parties}
			\rkVerb[%
			\rkPrep{with}{them},
			\rkSubClause[to]{%
				\rkVerb[%
				\rkAdv{ashore},
				\rkPrep[%
				\rkAdj{so~much},
				\rkAdj{a},
				\rkAppos{silver coin of any country}%
				]{at}{head}%
				]{take}
				\rkDObj{us}%
			}%
			]{contracted}
		\end{rkdiagram}
		\par\textit{(Mark Twain, \textit{The Innocents Abroad}, 1869)}
	}
	
\end{document}
